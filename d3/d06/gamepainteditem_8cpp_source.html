34GamePaintedItem::GamePaintedItem()
35{
36 if (m_pGame == nullptr)
37 m_pGame = new GameCore(QSize(m_SpSettingsProxy->getFieldWidth(),
38 m_SpSettingsProxy->getFieldHeight()),
39 m_SpSettingsProxy->getWinLength(),
40 m_SpSettingsProxy->getFirstPlayer(),
41 m_SpSettingsProxy->getPlayersCount());
42
43 this->setMipmap(true);
44 this->setAntialiasing(true);
45 setAcceptedMouseButtons(Qt::AllButtons);
46 m_nLocalPlayers.append(new QString("Game Over!"));
47 for (uint i = 1; i <= m_pGame->getNPlayers(); ++i) {
48 auto playerName = m_SpSettingsProxy->value("playersConf/" + QString::number(i) + "Name");
49 if (playerName.isNull()) {
50 m_SpSettingsProxy->setValue("playersConf/" + QString::number(i) + "Name", "Player" + QString::number(i));
51 m_SpSettingsProxy->getSpSettings()->sync();
52 playerName = m_SpSettingsProxy->value("playersConf/" + QString::number(i) + "Name");
53 }
54 m_nLocalPlayers.append(new QString(playerName.toString()));
55 }
56
57 m_nPlayersPic.append(new QImage(m_SpSettingsProxy->value("images/" + m_SpSettingsProxy->getPlayersConf("background")).value<QImage>()));
58 m_nWinPlayersPic.append(new QImage(m_SpSettingsProxy->value("images/" + m_SpSettingsProxy->getPlayersConf("over")).value<QImage>()));
59
60 for (uint i = 1; i <= m_SpSettingsProxy->getPlayersCount(); ++i) {
61 m_nPlayersPic.append(new QImage(
62 m_SpSettingsProxy->value("images/" + m_SpSettingsProxy->getPlayersConf(QString::number(i) + "Turn")).value<QImage>()));
63 m_nWinPlayersPic.append(new QImage(
64 m_SpSettingsProxy->value("images/" + m_SpSettingsProxy->getPlayersConf(QString::number(i) + "Win")).value<QImage>()));
65 }
66
67 connect(m_pGame,
68 SIGNAL(onEnd(QSize, QSize, uint)),
69 SLOT(onEnd(QSize, QSize, uint)));

71 SIGNAL(onMove(QSize, uint)),

133QString GamePaintedItem::gameState()
134{
135 QString val;
136 if (m_pEndValue == nullptr) {
137 val = m_SpSettingsProxy->getPlayersConf(QString::number(m_pGame->turn()) + "Name");
138 val.append(" turn..");
139 } else if (m_pEndValue->winPlayer > 0) {
140 // val = m_SpSettingsProxy->getPlayersConf(QString::number(m_pEndValue->winPlayer) + "Name" );
141 val = *m_nLocalPlayers[m_pEndValue->winPlayer];
142 val.append(" WON!");
143 } else
144 val = *m_nLocalPlayers[m_pEndValue->winPlayer];

164void GamePaintedItem::mouseReleaseEvent(QMouseEvent* event)
165{
166 if (m_pEndValue != nullptr)
167 return;
168 if (m_nLocalPlayers[m_pGame->turn()] == nullptr)
169 return;
170 if (0 > event->position().rx() / getCellWidth() || 0 > event->position().ry() / getCellHeight())
171 return;
172 if (m_mousePressPoint == QSize(event->position().rx() / getCellWidth(), event->position().ry() / getCellHeight()))
173 m_pGame->move(m_mousePressPoint, m_pGame->turn());
174}

219void GamePaintedItem::drawCells(QPainter* painter, const qreal& cellWidth, const qreal& cellHeight)
220{
221 painter->save();
222 for (uint i = 0; i < m_pGame->getWidth(); ++i) {
223 for (uint j = 0; j < m_pGame->getHeight(); ++j) {
224 if (m_pGame->getField(i, j) > 0) {
225 painter->setBrush(QBrush(m_nPlayersPic[m_pGame->getField(i, j)]->scaled(cellWidth, cellHeight)));
226 painter->drawRect(cellWidth * i, cellHeight * j,
227 cellWidth, cellHeight);
228 }
229 }
230 }
231 painter->restore();
232}

234void GamePaintedItem::drawEnd(QPainter* painter)
235{
236 if (m_pEndValue != nullptr) {
237 uint minAspectRatio = qMin(this->width(), this->height());
238 painter->save();
239 painter->setPen(QPen(Qt::transparent, 0));
240 painter->setBrush(QBrush(
241 m_nWinPlayersPic[m_pEndValue->winPlayer]->scaled(minAspectRatio, minAspectRatio)));
242 painter->translate(this->width() > minAspectRatio ? (this->width() - minAspectRatio) / 2 : 0,
243 this->height() > minAspectRatio ? (this->height() - minAspectRatio) / 2 : 0);
244 painter->drawRect(0, 0, minAspectRatio, minAspectRatio);
245 painter->restore();
246 }
