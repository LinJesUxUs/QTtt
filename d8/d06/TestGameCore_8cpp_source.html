115void TestGameCore::move()
116{
117 QFETCH(int, width);
118 QFETCH(int, height);
119 QFETCH(int, m1x);
120 QFETCH(int, m1y);
121 QFETCH(int, p1);
122 QFETCH(int, f1);
123 QFETCH(int, m2x);
124 QFETCH(int, m2y);
125 QFETCH(int, p2);
126 QFETCH(int, f2);
127 struct moves {
128 QSize position;
129 int player;
130 int expected;
131 };
132 QList<moves> listMoves = { { QSize(m1x, m1y), p1, f1 },
133 { QSize(m2x, m2y), p2, f2 } };
134 uint retVal;
135 GameCore* game = new GameCore(QSize(width, height), 3, listMoves[0].player);
136 QList<QVariant> args;
137 QSignalSpy spyMove(game, SIGNAL(onMove(QSize, uint)));
138 QSignalSpy spyNoMove(game, SIGNAL(onNoMove(QSize, uint)));
139
140 for (moves i : qAsConst(listMoves)) {
141 game->move(i.position, i.player);
142
143 try {
144 retVal = game->getField(i.position);
145 } catch (const char* err_msg) {
146 retVal = 0;
147 }
148 QCOMPARE(retVal, (uint)i.expected);
149
150 if (i.player == i.expected) {
151 QCOMPARE(spyMove.count(), 1);
152 args = spyMove.takeFirst();
153 } else {
154 QCOMPARE(spyNoMove.count(), 1);
155 args = spyNoMove.takeFirst();
156 }
157 QCOMPARE(args.at(0).toSize(), i.position);
158 QCOMPARE(args.at(1).toInt(), i.player);
159 }
160}

175void TestGameCore::getWinLength()
176{

178 QFETCH(int, height);
179 QFETCH(int, winLength);
180 GameCore game(QSize(width, height), winLength);
181 if (winLength < 1)

319 game->move(QSize(m4x, m4y), 2);
320 game->move(QSize(m5x, m5y), 1);
