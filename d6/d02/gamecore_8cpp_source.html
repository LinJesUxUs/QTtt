17GameCore::GameCore(const QSize& field, const uint& winLength, const uint& firstPlayer, const uint& players, QObject* parent)
18 : QObject(parent)
19{
20 m_nField.resize(field.width() < 1 ? 1 : field.width());
21 for (int i = 0; i < m_nField.size(); ++i)
22 m_nField[i].resize(field.height() < 1 ? 1 : field.height());
23
24 mWinLength = qMin(
25 int(int(winLength) < 1 ? 1 : winLength),
26 (int)qMin(m_nField.size(), m_nField[0].size()));
27
28 mPlayers = (int(players) < 0) ? 0 : players;
29 mTurn = firstPlayer > mPlayers ? mPlayers : firstPlayer;
30}

32void GameCore::isEnd(const QSize& pos) const
33{
34 QSize begin;
35 QSize end;
36 QList<QSize> lst(Directions::getList());
37 for (int i = 0; i < lst.size(); ++i) {
38 begin = vecLen(pos, Directions::negative(lst[i]));
39 end = vecLen(pos, lst[i]);
40 if (isWinRange(begin, end)) {
41 emit onEnd(begin, end, getField(pos));
42 return;
43 }
44 }
45}

47bool GameCore::isWinRange(const QSize& begin, const QSize& end) const
48{
49 if (begin.width() == end.width())
50 return (mWinLength == (uint)(end.height() - begin.height() + 1));
51 return (mWinLength == (uint)(begin.width() > end.width() ? begin.width() - end.width() : end.width() - begin.width()) + 1);
52}

54const QSize GameCore::vecLen(const QSize& pos, const QSize& offset) const
55{
56 QSize val = pos;
57 try {
58 uint fPos = getField(pos);
59 do {
60 val.rwidth() += offset.width();
61 val.rheight() += offset.height();
62 } while (fPos == getField(val));
63 } catch (const char* err_msg) {
64 }
65 val.rwidth() -= offset.width();
66 val.rheight() -= offset.height();
67 if (isOutOfRange(val)) {
68 val.rwidth() += offset.width();
69 val.rheight() += offset.height();
70 }
71 return val;
72}

165void GameCore::move(const QSize& pos, const uint& player)
166{
167 try {
168 if (!isBusy(pos)) {
169 if (player == mTurn) {
170 m_nField[pos.width()][pos.height()] = player;
171 isEnd(pos);
172 nextTurn();
173 if (++mTurnCount >= m_nField.size() * m_nField[0].size()) {
174 emit onEnd(QSize(), QSize(), 0);
175 }
176 emit onMove(pos, player);
177 return;
178 }
179 }
180 emit onNoMove(pos, player);
181 } catch (const char* err_msg) {
182 emit onNoMove(pos, player);
183 }
184}
